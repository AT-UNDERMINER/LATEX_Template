% IEEE section headings
%   \section:       "I. INTRODUCTION"  Roman numeral, centred, small capitals
%   \subsection:    "A. Heading"       capital letter, italic, flush left
%   \subsubsection: "1) Heading:"      run into the paragraph, italic, indented 1 em
% \cref gives "Section II", "Section II-A" and "Section II-A1"
\renewcommand{\thesection}{\Roman{section}}
\renewcommand{\thesubsection}{\thesection-\Alph{subsection}}
\renewcommand{\thesubsubsection}{\thesubsection\arabic{subsubsection}}
\titleformat{\section}{\normalfont\normalsize\scshape\centering}{\Roman{section}.}{0.5em}{}
\titleformat{\subsection}{\normalfont\normalsize\itshape}{\Alph{subsection}.}{0.5em}{}
\titleformat{\subsubsection}[runin]{\normalfont\normalsize\itshape}{\arabic{subsubsection})}{0.5em}{}[:]
\titlespacing*{\section}{0pt}{3ex plus 1ex minus .2ex}{1.5ex plus .2ex}
\titlespacing*{\subsection}{0pt}{2ex plus 1ex minus .2ex}{1ex plus .2ex}
\titlespacing*{\subsubsection}{1em}{1.5ex plus .5ex minus .2ex}{0.5em}

% The report class numbers floats per chapter (Figure 0.1). The main body has no
% chapters, so number figures, tables and equations plainly (Fig. 1).
\counterwithout{figure}{chapter}
\counterwithout{table}{chapter}
\counterwithout{equation}{chapter}

% IEEE caption style
%   Figures:    "Fig. 1. Caption text." below the figure
%   Subfigures: "(a) Caption text." and referenced as "Fig. 2(a)"
%   Tables:     "TABLE I" with the title in small capitals on the next line, above the table
\captionsetup[figure]{name=Fig., labelsep=period}
\renewcommand{\thesubfigure}{(\alph{subfigure})}
\captionsetup[sub]{labelformat=simple}
\renewcommand{\thetable}{\Roman{table}}
\captionsetup[table]{position=top, name=TABLE, labelsep=newline, textfont=sc, justification=centering}

% IEEE international date format: "7 October 2026" (used by \today on the title page)
\renewcommand{\today}{\number\day\space\ifcase\month\or January\or February\or
    March\or April\or May\or June\or July\or August\or September\or October\or
    November\or December\fi\space\number\year}

% IEEE uses "References" as the heading for the reference list
\renewcommand{\bibname}{References}
