\renewcommand{\citepunct}{], [} % IEEE style: [1], [3] rather than [1,3]

\usepackage[capitalise,nameinlink]{cleveref} % Smart references: \cref{fig:x} gives "Fig. 1" (must load after hyperref)
% IEEE reference style:
%   Figures:    "Fig. 1" everywhere, even at the start of a sentence; "Figs. 3--5" for ranges
%   Subfigures: "Fig. 2(a)", and the singular "Fig. 3(a)--3(d)" for parts of one figure
%   Equations:  "(1)" mid-sentence (\cref), "Equation (1)" at the start of a sentence (\Cref)
%   Tables:     "Table I" (Roman numerals are set in Format Parameters.tex)
\newcommand{\crefrangeconjunction}{--} % En dash for ranges: "Figs. 3--5", "(1)--(3)"
\newcommand{\creflastconjunction}{, and~} % Serial comma in lists: "Figs. 1, 3, and 5"
\crefname{figure}{Fig.}{Figs.}
\Crefname{figure}{Fig.}{Figs.}
\crefformat{subfigure}{#2Fig.~#1#3}
\Crefformat{subfigure}{#2Fig.~#1#3}
\crefmultiformat{subfigure}{#2Fig.~#1#3}{ and~#2#1#3}{, #2#1#3}{, and~#2#1#3}
\Crefmultiformat{subfigure}{#2Fig.~#1#3}{ and~#2#1#3}{, #2#1#3}{, and~#2#1#3}
\crefrangeformat{subfigure}{#3Fig.~#1#4--#5#2#6}
\Crefrangeformat{subfigure}{#3Fig.~#1#4--#5#2#6}
\crefname{table}{Table}{Tables}
\Crefname{table}{Table}{Tables}
\crefformat{equation}{#2(#1)#3}
\Crefformat{equation}{#2Equation~(#1)#3}
\crefmultiformat{equation}{#2(#1)#3}{ and~#2(#1)#3}{, #2(#1)#3}{, and~#2(#1)#3}
\Crefmultiformat{equation}{#2Equations~(#1)#3}{ and~#2(#1)#3}{, #2(#1)#3}{, and~#2(#1)#3}
\crefrangeformat{equation}{#3(#1)#4--#5(#2)#6}
\Crefrangeformat{equation}{#3Equations~(#1)#4--#5(#2)#6}
\crefname{lstlisting}{Listing}{Listings}     % Name used for code listings

% Code and Listings  
\usepackage{listings}   % Code inclusion (formats and highlights source code snippets)  

% SI Units and Measurements  
\usepackage{siunitx}    % Proper typesetting of units and numbers (handles SI units consistently)
% IEEE unit style
\sisetup{
    quantity-product     = \nobreakspace,                 % Regular space between number and unit: 10 V
    inter-unit-product   = \ensuremath{{}\cdot{}},        % Centre dot between units: N·m
    per-mode             = symbol,                        % Slash for "per": m/s, rad/s
    range-units          = single,                        % Unit once in ranges: 10--50 Hz
    range-phrase         = --,                            % En dash in ranges
    list-units           = single,                        % Unit once in lists: 1, 2, and 5 mm
    list-pair-separator  = {~and~},
    list-final-separator = {, and~},                      % Serial comma before "and"
    product-units        = single,                        % 2 x 5 cm
}
% Percent has no space and is repeated in ranges and lists: 20%--30%
\DeclareSIUnit[quantity-product = {}, range-units = repeat, list-units = repeat]{\percent}{\%}
